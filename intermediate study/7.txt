배열 메소드 (기초에서 한거)
push() : 뒤에 삽입
pop() : 뒤에 삭제
unshift() : 앞에 삽입
shift() : 앞에 삭제


arr.splice(n,m) : 특정 요소 지움
n번째 요소부터 m개를 지우라는 뜻


        let arr = [1,2,3,4,5];
        arr.splice(1,2);
        console.log(arr);

출력결과 : [1, 4, 5]



arr.splice(n,m,x) : 특정 요소 지우고 추가

        let arr = [1,2,3,4,5];
        arr.splice(1, 3, 100, 200);
        console.log(arr);

출력결과 : [1, 100, 200, 5]



arr.splice(n,m,x) 를 활용 (m 에 0을 대입하여 중간에 요소를 추가하기만 할수도 있음.)
        let arr = ["나는", "철수", "입니다"];
        arr.splice(1, 0, "대한민국", "소방관");
        console.log(arr);

출력결과 : 나는 대한민국 소방관 철수 입니다


arr.splice() : 삭제된 요소 반환

        let arr = [1,2,3,4,5];

        let result = arr.splice(1,2);

        console.log(arr);
        console.log(result);
출력결과 : [1,4,5]
	 [2,3]




arr.slice(n,m) : n부터 m까지 반환

        let arr = [1,2,3,4,5];
        console.log(arr.slice(1,4));

        let arr2 = arr.slice();
        console.log(arr2);

출력결과 : [2, 3, 4]
	 [1, 2, 3, 4, 5]







arr.concat(arr2, arr3 ..) : 합쳐서 새 배열 반환

        let arr = [1,2];

        arr = arr.concat([3,4]);

        console.log(arr);

출력결과 : [1, 2, 3, 4]





arr.forEach(fn) : 배열 반복

forEach(item, index, arr)
1. 해당요소
2. 인덱스
3. 해당 배열 자체 (생략가능)


        let users = ["Mike", "Tom", "Jane"];
        users.forEach((name, index) => {
          console.log(`${index + 1}. ${name}`);
        })

출력결과 : 1. Mike
	 2. Tom
	 3. Jane

arr.lastIndexOf : indexOf를 끝에서부터 탐색하는것.















arr.find(fn) / arr.findIndex(fn) : indexOf랑 같게 동작하는데, 함수를 넣어서 사용할 수 있음.
			 주의할점 : 첫번째 true 값만 반환하고 끝.
			 만약 없으면 indefined 를 반환




        let arr = [1, 2, 3, 4, 5];

        const result = arr.find((item) => { // ★
          return item % 2 === 0;
        });

        console.log(result);

출력결과 : 2




★ const result = arr.find(function(item) { 로 쓸 수도 있음. (함수표현식)

          { name : "Mike", age: 30},
          { name : "Jane", age: 27},
          { name : "Tom", age: 10},
        ];

        const result = userList.find((user) => {
          if (user.age < 19) {

arr.filter(fn) : 만족하는 모든 요소를 배열로 반환 (find랑 사용법이 같음.)

        const arr = [1,2,3,4,5,6];

        const result = arr.filter((item) => {
          return item % 2 === 0;
        });

        console.log(result);
출력결과 : [2, 4, 6]



arr.reverse() : 역순으로 재정렬

        let arr = [1,2,3,4,5];
        arr.reverse();
        console.log(arr);
출력결과 : [5, 4, 3, 2, 1]



★★★★★ 실무에서 많이 사용함 ★★★★★
arr.map(fn) : 함수를 받아 특정 기능을 시행하고 새로운 배열을 반환


        let userList = [
          { name: "Mike", age: 30},
          { name: "Jane", age: 27},
          { name: "Tom", age: 10},
        ];

        let newUserList = userList.map((user, index) => {
          return Object.assign({}, user, {
            id : index + 1,
            isAdult : user.age > 19,
          });
        });
      console.log(newUserList);

출력결과 : {name: "Mike", age: 30, id: 1, isAdult: true}
	 {name: "Jane", age: 27, id: 2, isAdult: true}
	 {name: "Tom", age: 10, id: 3, isAdult: false}





join : 배열을 합쳐서 문자열을 만듦.

        let arr = ["안녕", "나는", "철수야"];

        let result = arr.join(); // join 에 어떤 매개변수를 넣느냐에 따라 달라짐.
		          // join('  ') 이면 출력결과는 안녕 나는 철수야 로 바뀜.
        console.log(result);
출력결과 : 안녕,나는,철수야







split : 문자열을 나눠서 배열로 만듦.



        const users = "Mike,Jane,Tom,Tony";

	 ["H", "e", "l", "l", "o", ",", " ", "M", "y", " ", "n", "a", "m", "e", " ", "i", "s", " ", "M", "i", "k", "e"]





Array.isArray() : 배열인지 아닌지 확인

만약, typeof 연산자를 사용하면 둘다 Object 라고 출력돼서 뭐가 객체고, 뭐가 배열인지 알 수 없음.

        let user = {
          name : "Mike",
          age : 30,
        }

        let userList = ["Mike", "Tom", "Jane"];

        console.log(typeof user);
        console.log(typeof userList);

        console.log(Array.isArray(user));
        console.log(Array.isArray(userList));
출력결과 : Object
	 Object
	 false
	 true
