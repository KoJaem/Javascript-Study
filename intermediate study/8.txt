arr.sort() : 배열 재정렬, 배열 자체가 변경되니 주의


        let arr = [27, 8, 5, 13];

        arr.sort((a,b) => {
          return a-b;
        });

        console.log(arr);

출력결과 : [5, 8, 13, 27]

★
arr.sort() 를 하면 안되는 이유 : 아스키코드로 비교하기 때문에
만약 배열이 [1, 4, 20, 3] 이면 앞 숫자만 인식하기 때문에 [1, 20, 3, 4] 가 되어버기 때문!







arr.reduce() : 배열의값을 돌면서 원하는작업을 하고 최종값을 반환함.

예를들어 배열의 모든 수를 합치는 코드가 있다면

        let arr = [1, 2, 3, 4, 5];

        let result = 0;
        arr.forEach(num => {
          result += num;
        })

        console.log(result);
출력결과 : 15


이런식으로 할 수 있지만,




arr.reduce() 를 사용하면

        let arr = [1, 2, 3, 4, 5];

        const result = arr.reduce((prev, cur)=>{
          return prev + cur;
        }, 0) // 0은 초기값인데, 선택사항이므로 생략 가능. 만약 100으로 설정하면 이 코드에선 출력결과가 115
        console.log(result);
출력결과 : 15


arr.reduce(누적 계산값, 현재값) => { return 계산값 };





실용적인 예제
userList를 받아서 성인인지 아닌지 판단하기 (배열메소드 첫번째 공부할때도 map으로 비슷한 예제 했음!)

let userList = [
          { name: "Mike", age: 30},
          { name: "Tom", age: 10},
          { name: "Jane", age: 27},
          { name: "Sue", age: 26},
          { name: "Harry", age: 13},
          { name: "Steve", age: 60},
      ];

        let result = userList.reduce((prev, cur)=> {
          if(cur.age > 19) {
            prev.push(cur.name);
          }
          return prev;
        }, [])

        console.log(result);
출력결과 : ["Mike", "Jane", "Sue", "Steve"]

이름이 3글자인 사람만 출력

        let userList = [
          { name: "Mike", age: 30},
          { name: "Tom", age: 10},
          { name: "Jane", age: 27},
          { name: "Sue", age: 26},
          { name: "Harry", age: 13},
          { name: "Steve", age: 60},
      ];

        let result = userList.reduce((prev, cur)=> {
          if(cur.name.length === 3) {
            prev.push(cur.name);
          }
          return prev;
        }, []);

        console.log(result);
